ESP32 Starter Code
Flash this to your ESP32 — it registers itself with ARIA automatically:
#include <WiFi.h> #include <HTTPClient.h> #include <ArduinoJson.h> const char* ssid = "YOUR_WIFI"; const char* password = "YOUR_PASS"; const char* aria_url = "http://192.168.1.100:5000"; // ARIA server IP const char* device_name = "ESP32_Node1"; void setup() { Serial.begin(115200); WiFi.begin(ssid, password); while (WiFi.status() != WL_CONNECTED) delay(500); Serial.println("WiFi connected: " + WiFi.localIP().toString()); // Register with ARIA HTTPClient http; http.begin(String(aria_url) + "/api/iot/devices"); http.addHeader("Content-Type", "application/json"); String payload = "{\"name\":\"" + String(device_name) + "\",\"ip\":\"" + WiFi.localIP().toString() + "\",\"port\":80,\"type\":\"esp32\"}"; int code = http.POST(payload); Serial.println("Registered: " + String(code)); http.end(); } void loop() { // Read sensor (example: temperature) float temp = 25.0 + random(-5, 5); // Send to ARIA HTTPClient http; http.begin(String(aria_url) + "/api/iot/sensor/" + device_name); // ... send sensor data // Check for commands from ARIA http.begin(String(aria_url) + "/api/iot/command"); // ... handle commands delay(5000); // every 5s }
